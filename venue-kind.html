works-at function works-at(e)
typst · 24 lines
#let works-at(e) = {
  let v = quote(e.id)
  let rules = (
    "at ?X <- meta ?X \"venue\" " + v + "\n"
      + "at ?X <- meta ?X \"container\" ?S, venue-alias " + v + " ?S, not has-venue ?X\n"
      + "has-venue ?X <- meta ?X \"venue\" _\n"
  )
  let ids = helia.query(rules + "?X -: at ?X, not unlisted ?X").map(r => r.X).dedup()
  // by year, newest first (a reference has a `year`, a talk a `date`); within a year, newest first
  let dated = ids.map(id => {
    let x = helia.entry(id)
    (year: year-of(x), id: id)
  })
  let years = ()
  for y in dated.map(d => d.year).dedup().sorted(key: y => if y == none { 0 } else { -y }) {
    years.push((y, newest-first(dated.filter(d => d.year == y).map(d => d.id))))
  }
  for (y, group) in years {
    helia.group("venue-year", {
      heading(level: 1, if y == none { [Undated] } else { [#y] })
      for id in group { helia.transclude(id, expanded: false, numbered: false) }
    })
  }
}

#let venue = helia.kind(
  name: "venue",
  extends: "note",
  fields: (
    name: (type: "string", optional: true),           // its full name
    type: (type: "string", optional: true),           // conference, journal, workshop, series, preprints, book
    publisher: (type: "string", optional: true),
    external: (type: "string", optional: true),       // its website
    aliases: (type: "list", of: "string", optional: true),  // `container` strings that mean it
  ),
  relations: (
    venue-alias: ("aliases",),  // venue-alias(venue, "Journal of Functional Programming")
  ),
  ..template(
    byline: e => {
      let t = get(e, "type")
      by(get(e, "name"), if t != none { type-names.at(t, default: t) }, get(e, "publisher"))
    },
    links: (external: (label: v => [#bare-url(v)])),
    after: works-at,
    backlinks: false,   // what cites the venue is what appeared there, listed by year
  ),
  skeletons: (
    default: ```
      #import "venue-kind": venue
      #show: venue
      #helia.meta(title: [$TITLE], name: "", type: "conference")
      ```,
  ),
)
