#let course-summary(e) = {
  let has(k) = get(e, k) != none and get(e, k) != ()
  let line = by(
    get(e, "role"),
    if has("instructor") [Instructor #people(get(e, "instructor"))],
    if has("staff") [With #people(get(e, "staff"))],
  )
  if line != none { helia.group("course-role", line) }
  if has("description") { helia.group("abstract", get(e, "description")) }
}

  ..template(
    byline: e => by(get(e, "code"), get(e, "term"), get(e, "institution")),
    links: (external: [Course website]),
    summary: course-summary,
  ),
  skeletons: (
    default: ```
      #import "course-kind": course

      #helia.meta(title: [$TITLE], author: "$AUTHOR", code: "", term: "", institution: "", role: "")
      ```,
  ),
