      })
    }

#let person = helia.kind(
  name: "person",
  extends: "note",
  fields: (
    position: (type: "string", optional: true),       // "Assistant Professor"
    // Affiliations, each an institution entry's ID (or a name): where they are now, where they
    // studied, and where they were before. They connect people on institution pages.
    institution: (type: "string", optional: true),
    // (one institution, or several: a dual degree)
    phd-institution: (type: "list", of: "string", optional: true),
    masters-institution: (type: "list", of: "string", optional: true),
    undergrad-institution: (type: "list", of: "string", optional: true),
    former-institutions: (type: "list", of: "string", optional: true),
    external: (type: "string", optional: true),       // their website
    orcid: (type: "string", optional: true),
    email: (type: "string", optional: true),
    photo: (type: "asset-ref", optional: true),        // a portrait, on their own page
    github: (type: "string", optional: true),         // a user name
    mastodon: (type: "string", optional: true),       // the profile's URL
    cv: (type: "any", optional: true),                // a CV: a PDF asset or a URL
    // The name as BibTeX should read it, "Last, First" (a multi-part surname).
    cite-as: (type: "string", optional: true),
    phd-advisor: (type: "list", of: "string", optional: true),
    postdoc-advisor: (type: "list", of: "string", optional: true),
  ),
  relations: (
    phd-advisor: ("phd-advisor",),                       // phd-advisor(person, advisor)
    postdoc-advisor: ("postdoc-advisor",),               // postdoc-advisor(person, advisor)
    institution: ("institution",),                       // institution(person, institution)
    phd-institution: ("phd-institution",),               // phd-institution(person, institution)
    masters-institution: ("masters-institution",),       // (person, institution), likewise
    undergrad-institution: ("undergrad-institution",),   // (person, institution), likewise
    former-institution: ("former-institutions",),        // former-institution(person, institution)
  ),
  ..template(
    byline: e => by(get(e, "position"), place(get(e, "institution"))),
    // the addresses themselves; ORCID, GitHub and Mastodon get the theme's logos
    links: (
      external: (label: v => [#bare-url(v)]),
      email: (label: v => [#v], prefix: "mailto:"),
      orcid: (label: v => [#v], prefix: "https://orcid.org/"),
      github: (label: v => [#v], prefix: "https://github.com/"),
      mastodon: (label: v => [#handle(v)]),
      cv: [CV],
    ),
    figure: e => {
      let photo = get(e, "photo")
      if type(photo) == dictionary and photo.at("name", default: "") != "" {
        helia.group("person-photo", image(photo.name, alt: "Photo"))
      }
    },
    summary: connections,
    after: works,
  ),
  skeletons: (
    default: ```
      #import "person-kind": person
      #show: person
      #helia.meta(title: [$TITLE], position: "", institution: "", external: "")
      ```,
  ),
)
