#let tagged(e) = {
  let name = get(e, "name")
  if name == none { return }
  let rows = helia.query("?X, ?K, ?D -: tag ?X " + quote(name) + ", kind ?X ?K, not unlisted ?X, sort-date ?X ?D\norder by ?D desc, ?X")
  let groups = (
    ([References], ("reference",)),
    ([Talks and videos], ("talk", "video")),
    ([People], ("person",)),
  )
  let listed = ()
  for (title, kinds) in groups {
    let ids = rows.filter(r => r.K in kinds).map(r => r.X).dedup()
    listed += ids
    if ids.len() > 0 {
      helia.group("tag-group", {
        heading(level: 1, [#title (#ids.len())])
        for id in ids { helia.transclude(id, expanded: false, numbered: false) }
      })
    }
  }
  let rest = rows.map(r => r.X).dedup().filter(x => x not in listed and x != e.id)
  if rest.len() > 0 {
    helia.group("tag-group", {
      heading(level: 1, [Notes (#rest.len())])
      for id in rest { helia.transclude(id, expanded: false, numbered: false) }
    })
  }
}
