#let rule(name: none, label: none, ..args) = {
  let parts = args.pos().map(_src)
  if name != none { parts.push("name: " + _name(name)) }
  if label != none { parts.push("label: " + _name(label)) }
  (curryst: "rule(" + parts.join(", ") + ")")
}

derivation function derivation(..items)
typst · 27 lines
#let derivation(..items) = {
  let cells = ()
  let rows = ()
  let bar = "grid.hline(stroke: 0.4pt, end: 1)"
  for it in items.pos() {
    if type(it) == dictionary and "step" in it {
      let s = it.step
      let name = if s.name == none { "[]" } else {
        "grid.cell(inset: 0pt, box(height: 0pt, align(horizon, pad(left: 0.4em, " + _name(s.name) + "))))"
      }
      cells.push(bar)
      cells.push("[]")
      cells.push(name)
      if s.invertible {
        cells.push(bar)
        rows.push("2pt")
      } else {
        rows.push("0pt")
      }
    } else {
      cells.push(_src(it))
      cells.push("[]")
      rows.push("auto")
    }
  }
  _island("#grid(columns: 2, align: (center, left), rows: (" + rows.join(", ") + "), inset: (x: 0.3em, y: 0.25em), " + cells.join(", ") + ")")
}
