#let home-layout(entry, body) = {
  let m = entry.meta
  let person = helia.entry(m.at("person", default: ""))
  let name = if person != none { person.title } else { entry.title }
  let photo = m.at("photo", default: none)
  page-layout(entry, {
    masthead(helia.setting("site-name"))
    helia.group("home-hero", {
      if photo != none {
        let attrs = ("src": photo.url, "alt": helia.setting("site-name"), "class": "helia-home-photo")
        el("img", attrs + ("width": "168", "height": "168"), [])
      }
      helia.group("home-intro", {
        el("h1", ("class": "helia-home-name"), name)
        body
      })
      if person != none { contact-links(person.meta) }
    })
    for s in m.at("sections", default: ()) { helia.transclude(s, section: true) }
  })
}
