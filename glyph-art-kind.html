#let glyph-art = helia.kind(
  name: "glyph-art",
  extends: "note",
  fields: (
    brushes: (type: "list", of: "string", optional: true),
    rows: (type: "list", of: "string", optional: true),
    cell: (type: "number", optional: true),
  ),
  layout: (entry, body) => page-layout(entry, body, extra: art(entry)),
  transcluded: (entry, body) => { art(entry); body },
  skeletons: (
    default: ```
      #import "glyph-art-kind": glyph-art
      #show: glyph-art
      #helia.meta(title: [$TITLE], brushes: ("r = ⊕ ⊗ : accent",), rows: (".....", "..r..", "....."))

      $CURSOR
      ```,
  ),
)
