#let authors-marking-speaker(e) = {
  let a = get(e, "author")
  let s = get(e, "speaker")
  let a = if a == none { () } else { a }
  let s = if s == none { () } else { s }
  let people = s.filter(p => p not in a) + a
  if people.len() > 0 {
    people.map(p => if p in s { helia.span("speaker", strong(helia.link(p))) } else { helia.link(p) }).join([, ])
  }
}

#let talk = helia.kind(
  name: "talk",
  extends: "note",
  fields: (
    speaker: (type: "list", of: "string", optional: true),  // who gave it (people's IDs)
    venue: (type: "string", optional: true),         // where it was given ("PLDI")
    event: (type: "string", optional: true),         // the event's page
    slides: (type: "any", optional: true),           // a PDF asset or a URL
    video: (type: "string", optional: true),         // a recording (YouTube, Vimeo)
    abstract: (type: "content", optional: true),
    // the byline lists the authors itself, with the speaker marked
    show-author: (type: "boolean", default: false),
  ),
  relations: (
    speaker: ("speaker",),   // speaker(talk, person)
  ),
  ..template(
    byline: e => by(authors-marking-speaker(e), venue-link(get(e, "venue")), long-date(get(e, "date"))),
    links: (event: [Event], slides: [Slides], video: [Video]),
    summary: abstract-of,
    // the slides (a PDF) and the recording, each folding
    after: e => {
      pdf-viewer(e, field: "slides", summary: "Slides")
      if get(e, "video") != none { video(get(e, "video")) }
    },
  ),
  skeletons: (
    default: ```
      #import "talk-kind": talk
      #show: talk
      #helia.meta(title: [$TITLE], author: ("$AUTHOR",), date: "$DATE", venue: "", abstract: [])
      ```,
  ),
)
