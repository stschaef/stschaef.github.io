#let video = helia.kind(
  name: "video",
  extends: "note",
  fields: (
    venue: (type: "string", optional: true),         // what it was made for
    video: (type: "string", optional: true),         // where it is (YouTube, Vimeo)
    abstract: (type: "content", optional: true),
  ),
  ..template(
    byline: e => by(venue-link(get(e, "venue")), long-date(get(e, "date"))),
    links: (video: [Video]),
    summary: abstract-of,
    after: e => if get(e, "video") != none { player(get(e, "video")) },
  ),
  skeletons: (
    default: ```
      #import "video-kind": video
      #show: video
      #helia.meta(title: [$TITLE], author: ("$AUTHOR",), date: "$DATE", video: "", abstract: [])
      ```,
  ),
)
