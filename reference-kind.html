#let outside(doi, arxiv, title, year) = {
  let url = if doi != "" { "https://doi.org/" + doi } else if arxiv != "" {
    "https://arxiv.org/abs/" + arxiv
  }
  let label = if title != "" { title } else if doi != "" { doi } else { arxiv }
  let text = if url != none { link(url, label) } else { label }
  if year != 0 [#text (#year)] else [#text]
}
cites function cites(e)
typst · 22 lines
#let cites(e) = {
  let q = quote(e.id)
  let here = helia.query("?B -: bibliography-match " + q + " _ ?B").map(r => r.B)
  let listed = related("cites-listed", e.id)
  let here = newest-first((here + listed).dedup().filter(x => x != e.id))
  let away = helia.query(
    "?N, ?D, ?A, ?T, ?Y -: bibliography-item " + q + " ?N ?D ?A ?T ?Y, bibliography-unmatched " + q
      + " ?N\norder by ?Y desc, ?N",
  )
  let n = here.len() + away.len()
  if n > 0 {
    helia.group("reference-cites", {
      folded("reference-cites-here", "With notes", here)
      if away.len() > 0 {
        helia.group("reference-cites-away", {
          list(..away.map(r => outside(r.D, r.A, r.T, r.Y)))
        }, collapsible: "true", open: "true", summary: "External (" + str(away.len()) + ")")
      }
    }, collapsible: "true", open: "true",
      summary: "Cites " + str(n) + " works (" + str(here.len()) + " here)")
  }
}
reference kind
typst · 50 lines
#let reference = helia.kind(
  name: "reference",
  extends: "note",
  fields: reference-fields,
  relations: reference-relations,
  ..template(
    // "PLDI · 2025": the short `venue` when there is one, else the `container`
    byline: e => {
      let venue = get(e, "venue")
      by(if venue != none { venue-link(venue) } else { get(e, "container") }, get(e, "year"))
    },
    links: (
      pdf: [PDF],
      // an open copy elsewhere (pldb's open-access link), when ours is not published
      "pdf-url": (label: [PDF], unless: "pdf"),
      doi: (label: [DOI], prefix: "https://doi.org/"),
      url: (label: [Web], unless: "doi"),                     // the DOI reaches the publisher
      eprint: (label: [arXiv], prefix: "https://arxiv.org/abs/"),
      // another edition of the work (a PDF asset or a URL): a transcription, a translation
      transcription: [Transcription],
      translation: [Translation],
      code: [Source code],
      artifact: [Artifact],
      pldb: [pldb],
    ),
    summary: abstract-of,
    // the paper, how to cite it, what links here, what cites it here, and what it cites
    after: e => {
      pdf-viewer(e)
      cite-section(e)
      let cited-by = citing(e)
      backlinks-section(e, except: cited-by)
      folded("reference-cited-by", "Cited by", cited-by)
      cites(e)
    },
    backlinks: false,
    // one shared broad tag (`typetheory`) is a coincidence between papers; ask for two
    related-min: 2,
    role: "reference",
  ),
  skeletons: (
    default: ```
      #import "reference-kind": reference
      #show: reference
      #helia.meta(title: [$TITLE], entry-type: "article", author: ("$AUTHOR",), year: 2026)

      $CURSOR
      ```,
  ),
)
