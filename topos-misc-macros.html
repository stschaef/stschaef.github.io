  let r = calc.rem(m, 60)
  if h == 0 { str(m) + " min" } else if r == 0 { str(h) + " h" } else { str(h) + " h " + str(r) + " min" }
}

recipe-facts function recipe-facts(entry, compact: false)
typst · 17 lines
#let recipe-facts(entry, compact: false) = {
  let m = entry.meta
  let fact(label, value) = helia.span("recipe-fact", {
    helia.span("recipe-fact-label", label)
    [ ]
    helia.span("recipe-fact-value", value)
  })
  let items = ()
  if m.prep > 0 { items.push(fact([Prep], fmt-minutes(m.prep))) }
  if m.cook > 0 { items.push(fact([Cook], fmt-minutes(m.cook))) }
  if m.prep > 0 and m.cook > 0 { items.push(fact([Total], fmt-minutes(m.prep + m.cook))) }
  let servings = m.at("servings", default: none)
  if servings != none { items.push(fact([Serves], str(servings))) }
  let cuisine = m.at("cuisine", default: none)
  if cuisine != none and not compact { items.push(fact([Cuisine], cuisine)) }
  if items.len() > 0 { helia.group("recipe-facts", items.join([ ]), compact: compact) }
}

  let src = m.at("source", default: none)

  page-layout(
    entry,
    helia.group("recipe", {
      helia.group("recipe-ingredients", {
        [= Ingredients]

        ingredient-list(entry)
      }, collapsible: "true")
      // no Method heading over a recipe whose steps were never written down
      if body != none and repr(body) != "[]" {
        helia.group("recipe-method", {
          [= Method]
          body
        })
      }
    }),
    extra: {
      recipe-facts(entry)
      if src != none { helia.group("recipe-source", [Adapted from #src.]) }
    },
  )
}
